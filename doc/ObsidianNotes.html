def filter(notes, *terms)
  return notes if terms.empty?

  direct_matches = notes.select { |note| note.matches?(*terms) }
  return direct_matches if direct_matches.any?

  notes.select { |note| note.content_matches?(*terms) }
end

def find_by_ref(ref, notes)
  if ref =~ /^\d+$/
    idx = ref.to_i - 1
    return notes[idx] if idx >= 0 && idx < notes.length
  end

  normalized = ref.sub(/\.md\z/, '')

  notes.find do |note|
    candidates = [
      note.relative_path,
      note.relative_without_ext,
      File.basename(note.relative_path),
      File.basename(note.relative_without_ext),
    ]

    candidates.include?(ref) || candidates.include?(normalized)
  end
end

def notes(scopes: nil)
  paths =
    if scopes.nil? || scopes.empty?
      Dir.glob(File.join(cld_dir, '**', '*.md'))
    else
      scopes.flat_map do |scope|
        dirname = SCOPE_DIRS.fetch(scope)
        Dir.glob(File.join(cld_dir, dirname, '**', '*.md'))
      end
    end

  paths.uniq.sort_by { |path| sort_key(path) }.map do |path|
    relative = Pathname.new(path).relative_path_from(Pathname.new(vault_dir)).to_s
    Note.new(
      vault_path: path,
      vault_relative_path: relative,
      title: note_title(path),
      vault_name: vault_name
    )
  end
end

def pick(notes, hiiro:)
  return nil if notes.empty?
  return notes.first if notes.length == 1

  unless $stdin.tty? && $stdout.tty?
    raise AmbiguousSelectionError, format_matches(notes)
  end

  mapping = notes.each_with_index.each_with_object({}) do |(note, idx), acc|
    acc[note.display_string(idx)] = note
  end

  selected = hiiro.fuzzyfind_from_map(mapping)
  raise NoSelectionError if selected.nil?

  selected
end

def save_task_statuses(tasks)
  lines = File.readlines(task_index_path)

  tasks.each do |task|
    next unless task.toggled?
    next if task.done_line_index.nil?

    lines[task.done_line_index] = lines[task.done_line_index].sub(/\[( |x)\] Done/, task.done ? '[x] Done' : '[ ] Done')
    task.original_done = task.done
  end

  File.write(task_index_path, lines.join)
end

def tasks
  lines = File.readlines(task_index_path, chomp: true)
  in_tasks = false
  current = nil
  entries = []

  lines.each_with_index do |line, idx|
    case line
    when /^## Claude Tasks$/
      in_tasks = true
      next
    when /^## /
      if in_tasks
        entries << current if task_complete?(current)
        break
      end
    end

    next unless in_tasks

    if line.start_with?('### ')
      entries << current if task_complete?(current)
      current = Task.new(title: line.sub(/^### /, '').strip, cld_relative_dir: cld_relative_dir)
      next
    end

    next unless current

    stripped = line.strip

    case stripped
    when '- [ ] Done'
      current.done = false
      current.original_done = false if current.original_done.nil?
      current.done_line_index = idx
    when '- [x] Done'
      current.done = true
      current.original_done = true if current.original_done.nil?
      current.done_line_index = idx
    when /\A-\s+\*\*Task note:\*\* .*?\(([^)]+)\)/
      current.note_path = Regexp.last_match(1)
    when /\A-\s+\*\*Key:\*\* (.+)\z/
      current.key = Regexp.last_match(1).strip
    when /\A-\s+\*\*Project:\*\* (.+)\z/
      current.project = Regexp.last_match(1).strip
    when /\A-\s+\*\*Priority:\*\* (.+)\z/
      current.priority = Regexp.last_match(1).strip
    end
  end

  entries << current if task_complete?(current)
  entries
rescue Errno::ENOENT
  []
end
