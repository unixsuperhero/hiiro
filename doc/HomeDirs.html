    module HomeDirs

    Tracked, migration-aware T3CODE_HOME dirs.

    Each home dir lives at ~/.local/share/codehiiro/<YmdHMS>.<sig>, where <sig> is a 12-char hash of the app’s migration list (id + name pairs from migrationEntries in apps/server/src/persistence/Migrations.ts).

    The DB inside a home dir ($HOME/{userdata,dev}/state.sqlite) records which migrations actually ran, in effect_sql_migrations. A home dir is REUSABLE for an app if every applied (id, name) in its DB matches the app’s own (id -> name) map exactly (DB is a subset of the app). In that case the app forward-migrates the rest on startup. Any divergence (an id the app doesn’t have, or the same id under a different name) means we mint a fresh dir.

    Tracked in ~/.config/codehiiro/homedirs.yml alongside projects.yml / worktrees.yml.

def app_bundle_migrations(app_path)
  lines = asar_bin_lines(app_path)
  start = lines.index { |l| l =~ /const migrationEntries\s*=\s*\[/ }
  return [] unless start

  migs  = []
  i     = start + 1
  depth = 1
  while i < lines.size && depth.positive?
    line   = lines[i]
    depth += line.count('[') - line.count(']')
    if line =~ /^\s*(\d+),\s*$/
      id   = Regexp.last_match(1).to_i
      nxt  = lines[i + 1]
      migs << [id, Regexp.last_match(1)] if nxt && nxt =~ /^\s*"([^"]+)"/
    end
    i += 1
  end
  migs
end

    Parse [id, “Name”, …] rows from migrationEntries in a built .app bundle. The bundler splits each row across lines, so we walk the array block and pair an ‘<id>,` line with the `“Name”,` line right after it.

def app_migrations(repo)
  file = File.join(repo, MIGRATIONS_REL)
  return [] unless File.file?(file)

  File.readlines(file, chomp: true).filter_map do |line|
    m = line.match(/^\s*\[\s*(\d+)\s*,\s*"([^"]+)"/)
    [m[1].to_i, m[2]] if m
  end
end

    Parse [id, “Name”, …] rows out of migrationEntries. Ignores everything else (imports, comments) — only array rows match.

def asar_bin_lines(app_path)
  asar = File.join(app_path, 'Contents/Resources/app.asar')
  return [] unless File.file?(asar)

  Dir.mktmpdir('chcompat') do |dir|
    ok  = system('npx', '--yes', '@electron/asar', 'extract-file', asar, BUNDLE_MIGRATIONS_REL,
                 chdir: dir, out: File::NULL, err: File::NULL)
    out = File.join(dir, File.basename(BUNDLE_MIGRATIONS_REL))
    ok && File.file?(out) ? File.readlines(out, chomp: true) : []
  end
end

    Extract just bin.mjs out of the app’s asar and return its lines. Uses a block-scoped temp dir so nothing is left behind. [] if the .app / asar / entry is missing or extraction fails.

def compatible?(home, appmap)
  home_dbs(home).all? do |db|
    db_migrations(db).all? { |id, name| appmap[id] == name }
  end
end

def db_migrations(dbfile)
  return [] unless File.file?(dbfile)

  sql = 'SELECT migration_id, name FROM effect_sql_migrations ORDER BY migration_id;'
  out = Hiiro::Shell.capture_output('sqlite3', '-readonly', '-noheader', '-separator', "\t", dbfile, sql)
  out.to_s.each_line.filter_map do |l|
    id, name = l.chomp.split("\t", 2)
    [id.to_i, name] if id && name && !id.empty?
  end
rescue StandardError
  []
end

    Applied migrations recorded in a state.sqlite. [] if the file or table is missing / unreadable.

def resolve(repo, create: true)
  migs   = app_migrations(repo)
  sig    = signature(migs)
  appmap = migs.to_h

  records = load_records
  reusable = records
    .select { |r| Dir.exist?(r['path'].to_s) }
    .sort_by { |r| r['created'].to_s }
    .reverse
    .find { |r| compatible?(r['path'], appmap) }

  if reusable
    if create
      reusable['last_used'] = Time.now.strftime('%Y%m%dT%H%M%S')
      save_records(records)
    end
    return [reusable['path'], :reused, sig]
  end

  return [nil, :would_create, sig] unless create

  stamp = Time.now.strftime('%Y%m%dT%H%M%S')
  home  = File.join(DATA_ROOT, "#{stamp}.#{sig}")
  FileUtils.mkdir_p(home)
  records << {
    'path'       => home,
    'sig'        => sig,
    'created'    => stamp,
    'last_used'  => stamp,
    'repo'       => repo,
    'migrations' => migs.length,
    'max_id'     => (migs.map(&:first).max || 0),
  }
  save_records(records)
  [home, :created, sig]
end

    Resolve the T3CODE_HOME to use for ‘repo`. Reuses the newest compatible tracked dir, else (when create:) mints and records a new one. Returns [home_or_nil, :reused | :created | :would_create, sig].

def signature(migs)
  Digest::SHA256.hexdigest(migs.map { |id, name| "#{id}_#{name}" }.join("\n"))[0, 12]
end
