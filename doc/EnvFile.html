def all
  @all ||= lines.reject { |line| line.match?(/^\s*#/) || !line.match?(/[=:]/) }.each_with_object({}) { |line, h|
    name, val = line.split(/\s*[=:]\s*/, 2)
    next if name.match?(/TIMEOUT/)
    h[name&.strip] = val&.strip
  }
end
