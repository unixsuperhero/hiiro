def self.from_porcelain_block(lines)
  attrs = {}
  lines.each do |line|
    case line
    when /^worktree (.+)/
      attrs[:path] = $1
    when /^HEAD (.+)/
      attrs[:head] = $1
    when /^branch refs\/heads\/(.+)/
      attrs[:branch] = $1
    when 'detached'
      attrs[:detached] = true
    when 'bare'
      attrs[:bare] = true
    end
  end
  new(**attrs) if attrs[:path]
end
