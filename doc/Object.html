def append_gitignore(repo_root, patterns)
  path = File.join(repo_root, '.gitignore')
  content = File.exist?(path) ? File.binread(path) : ''
  existing_patterns = content.lines.map(&:chomp)
  new_patterns = patterns - existing_patterns
  return [path, []] if new_patterns.empty?

  File.open(path, 'ab') do |file|
    file.write("\n") unless content.empty? || content.end_with?("\n")
    new_patterns.each { |pattern| file.puts(pattern) }
  end

  [path, new_patterns]
end

def branch_query(args)
  opts = Hiiro::Options.parse(args) { flag(:all, short: :a, desc: 'show all rows (no limit)') }
  query_args = opts.args
  db = Hiiro::DB.connection
  limit = opts.all ? nil : 50

  if query_args.empty?
    ds = db[:branches]
    ds = ds.limit(limit) if limit
    rows = ds.all
  elsif query_args[0].include?(' ') || query_args[0].upcase =~ /\A(SELECT|INSERT|UPDATE|DELETE|WITH)/
    sql = query_args.join(' ')
    rows = db.fetch(sql).all
    rows = rows.first(limit) if limit
  else
    conditions = query_args.each_with_object({}) do |pair, h|
      k, v = pair.split('=', 2)
      h[k] = v if k && v
    end
    ds = db[:branches]
    conditions.each { |k, v| ds = ds.where(k.to_sym => v) }
    ds = ds.limit(limit) if limit
    rows = ds.all
  end

  return puts "(no results)" if rows.empty?
  keys = rows.first.keys
  col_widths = keys.map { |k| [k.to_s.length, rows.map { |r| r[k].to_s.length }.max].max.clamp(0, 60) }
  header = keys.each_with_index.map { |k, i| k.to_s.ljust(col_widths[i]) }.join("  ")
  puts header
  puts "-" * header.length
  rows.each do |row|
    puts keys.each_with_index.map { |k, i|
      val = row[k].to_s; val = val[0..57] + "..." if val.length > 60; val.ljust(col_widths[i])
    }.join("  ")
  end
  puts "(#{rows.length} rows)"
end

def cmd_add
  tf = Tempfile.new(['h-loop-add-', '.yml'])
  tf.write(ADD_TEMPLATE)
  tf.flush
  tf.close
  system(ENV['EDITOR'] || ENV['VISUAL'] || 'nvim', tf.path)

  raw = File.read(tf.path)
  tf.unlink
  cfg = YAML.safe_load(raw) || {}

  name = cfg['name'].to_s.strip
  if name.empty? || name == 'my-loop'
    warn 'aborted: name is empty or unchanged from template'
    return
  end
  if RESERVED.include?(name) && !BUILTINS.include?(name)
    warn "aborted: '#{name}' is reserved"
    return
  end
  if cfg['prompt'].to_s.strip.empty?
    warn 'aborted: prompt is empty'
    return
  end

  FileUtils.mkdir_p(loop_dir(name))
  File.write(config_path(name), YAML.dump(cfg))
  puts "added: #{name} -> #{config_path(name)}"
end

def config_entries(agent, path)
  return {} unless path && File.exist?(path)
  if agent == 'codex'
    entries = {}
    File.read(path).scan(/^\[mcp_servers\.(?:\"([^\"]+)\"|([^\]]+))\]\s*\n(.*?)(?=^\[|\z)/m) do |quoted, bare, body|
      url = body[/^\s*url\s*=\s*"([^"]+)"/, 1]
      entries[quoted || bare] = url if url
    end
    entries
  else
    _, keys, = AGENT_TARGETS[agent].call(Dir.home)
    data = JSON.parse(File.read(path))
    container = (keys || ['mcpServers']).inject(data) { |h, k| h.is_a?(Hash) ? h[k] : nil } || {}
    container.transform_values { |v| v.is_a?(Hash) ? (v['url'] || v['httpUrl'] || v['serverUrl']) : nil }.compact
  end
rescue JSON::ParserError
  {}
end

def tmux_capture(*args)
  out = IO.popen(['tmux', *args, err: [:child, :out]], &:read)
  [out.strip, $?.success?]
end

def session_exists?
  system('tmux', 'has-session', '-t', SESSION, out: File::NULL, err: File::NULL)
end

def ensure_session
  return if session_exists?
  system('tmux', 'new-session', '-d', '-s', SESSION, '-n', '_home')
end

def window_exists?(window_id)
  return false if window_id.nil? || window_id.empty?
  out, ok = tmux_capture('list-windows', '-a', '-F', '#{window_id}')
  ok && out.split("\n").include?(window_id)
end

def kill_window(window_id)
  return unless window_id && !window_id.empty?
  system('tmux', 'kill-window', '-t', window_id, out: File::NULL, err: File::NULL)
end

def new_window(name)
  ensure_session
  # -P prints info about the new window; -F formats it. We want its window_id.
  window_id, _ = tmux_capture('new-window', '-d', '-t', "#{SESSION}:", '-n', name, '-P', '-F', '#{window_id}')
  pane_id, _   = tmux_capture('list-panes', '-t', window_id, '-F', '#{pane_id}')
  [window_id, pane_id.split("\n").first]
end

def send_line(pane_id, line)
  system('tmux', 'send-keys', '-t', pane_id, line, 'Enter')
end

def paste_text(pane_id, text)
  tf = Tempfile.new(['h-loop-paste-', '.txt'])
  tf.write(text)
  tf.flush
  tf.close
  system('tmux', 'load-buffer', '-b', 'hloop', tf.path)
  # -d deletes the buffer after paste. -p disables bracketed paste.
  system('tmux', 'paste-buffer', '-d', '-p', '-b', 'hloop', '-t', pane_id)
ensure
  tf&.unlink
end

def start_loop(name)
  cfg = load_config(name)
  unless cfg
    warn "no config: #{name}"
    return false
  end
  cfg['name'] ||= name

  info = load_info
  existing = info.dig('loops', name)
  if existing && window_exists?(existing['window_id'])
    puts "already running: #{name} (#{existing['window_id']} #{existing['pane_id']})"
    return true
  end

  window_id, pane_id = new_window(name)
  send_line(pane_id, CLAUDE_CMD)
  sleep 2.5 # wait for claude to boot

  paste_text(pane_id, compile_prompt(cfg))
  sleep 0.3
  # Leave Vim insert mode before submitting the pasted multiline prompt.
  system('tmux', 'send-keys', '-t', pane_id, 'Escape')
  sleep 0.5
  system('tmux', 'send-keys', '-t', pane_id, 'Enter')

  info['session'] = SESSION
  info['loops'][name] = {
    'window_id'  => window_id,
    'pane_id'    => pane_id,
    'started_at' => Time.now.iso8601,
    'frequency'  => cfg['frequency'],
  }
  save_info(info)
  puts "started: #{name} (#{window_id} #{pane_id})"
  true
end

def stop_loop(name)
  info = load_info
  entry = info.dig('loops', name)
  if entry
    kill_window(entry['window_id'])
    info['loops'].delete(name)
    save_info(info)
    puts "stopped: #{name}"
  else
    puts "not running: #{name}"
  end
end

    title = task == '_untagged' ? '(no task)' : task

def extract_messages(records)
  messages = []
  current_msg_id = nil
  current_msg = nil

  records.each do |record|
    next if !record.is_a?(Hash)

    rec_type = record['type']
    next unless ['user', 'assistant'].include?(rec_type)

    message = record['message'] || {}
    next unless message.is_a?(Hash)

    msg_id = message['id']

    if rec_type == 'assistant' && msg_id && msg_id == current_msg_id
      new_content = message['content'] || []
      if current_msg && new_content.any?
        existing_types = {}
        current_msg['content'].each_with_index do |c, i|
          next unless c.is_a?(Hash)
          key = [c['type'], c['id'] || c['tool_use_id'] || '']
          existing_types[key] = i
        end

        new_content.each do |block|
          next unless block.is_a?(Hash)
          block_key = [block['type'], block['id'] || block['tool_use_id'] || '']
          if existing_types[block_key] && ['text', 'thinking'].include?(block['type'])
            idx = existing_types[block_key]
            current_msg['content'][idx]['text'] = block['text'] if block['type'] == 'text'
            current_msg['content'][idx]['thinking'] = block['thinking'] if block['type'] == 'thinking'
          elsif !existing_types[block_key]
            current_msg['content'] << block
            existing_types[block_key] = current_msg['content'].length - 1
          end
        end
      end
      next
    end

    messages << current_msg if current_msg
    current_msg = {
      'type' => rec_type,
      'role' => message['role'] || rec_type,
      'content' => message['content'] || [],
      'timestamp' => record['timestamp'],
      'cwd' => record['cwd'],
      'gitBranch' => record['gitBranch'],
      'model' => message['model'],
      'uuid' => record['uuid'],
    }
    current_msg_id = rec_type == 'assistant' ? msg_id : nil
  end

  messages << current_msg if current_msg
  messages
end

def glob_path(glob, bname=nil, opts:, skill: false)

  fulltext = opts.full

  filters ||= []
  filters = ['.*'] if filters.empty?

    [

    $stderr.puts "PATH: #{path}" if opts.verbose || opts.veryverbose

    re_s = filters.join(?|)
    re = Regexp.new(re_s)
    fs.each do |f|

      cmd = ['egrep', *more_args, '-m1', re_s, f.to_s]

          system(*cmd, **kwcmd)
        else
          base_f.to_s.match?(re)
        end

      if matches
        puts display_f
      else
        $stderr.puts "SKIPPING #{base_f}" if opts.veryverbose
      end
    end
  end

def render_by_task(all_status)

  all_status.each do |name, s|

    (s['action_items'] || []).each  { |it| grouped[it['task'] || '_untagged'][:actions]   << it.merge('_loop' => name) }

  end
  return '' if grouped.empty?

  cards = grouped.sort_by { |k, _| k == '_untagged' ? 'zzz' : k }.map do |task, bucket|
    title = task == '_untagged' ? '(no task)' : task
    parts = []
    unless bucket[:items].empty?

def human_duration(total_secs)
  total = total_secs.to_f
  sign = total.negative? ? '-' : ''

  parts << "#{hours}h"   if hours > 0

  parts << "#{secs}s"    if secs > 0
  sign + (parts.empty? ? '0s' : parts.join)

def ignore_candidate_groups(files)
  extensions = files.each_with_object([]) do |path, patterns|
    extension = File.extname(path)
    patterns << "*#{extension}" unless extension.empty? || extension == '.'
  end.uniq.sort

    'Directories' => directories
  }

def new_window(name)
  ensure_session

def compile_prompt(cfg)

  <<~MSG.strip
    /loop #{freq} #{prompt}

def start_loop(name)
  cfg = load_config(name)
  unless cfg
    warn "no config: #{name}"
    return false
  end
  cfg['name'] ||= name

  info = load_info
  existing = info.dig('loops', name)
  if existing && window_exists?(existing['window_id'])
    puts "already running: #{name} (#{existing['window_id']} #{existing['pane_id']})"
    return true
  end

  window_id, pane_id = new_window(name)
  send_line(pane_id, CLAUDE_CMD)
  sleep 2.5 # wait for claude to boot

  paste_text(pane_id, compile_prompt(cfg))
  sleep 0.3
  # Leave Vim insert mode before submitting the pasted multiline prompt.
  system('tmux', 'send-keys', '-t', pane_id, 'Escape')
  sleep 0.5
  system('tmux', 'send-keys', '-t', pane_id, 'Enter')

  info['session'] = SESSION
  info['loops'][name] = {
    'window_id'  => window_id,
    'pane_id'    => pane_id,
    'started_at' => Time.now.iso8601,
    'frequency'  => cfg['frequency'],
  }
  save_info(info)
  puts "started: #{name} (#{window_id} #{pane_id})"
  true
end

def stop_loop(name)
  info = load_info
  entry = info.dig('loops', name)
  if entry
    kill_window(entry['window_id'])
    info['loops'].delete(name)
    save_info(info)
    puts "stopped: #{name}"
  else
    puts "not running: #{name}"
  end
end

def cmd_add

  tf.flush

    return

  File.write(config_path(name), YAML.dump(cfg))
  puts "added: #{name} -> #{config_path(name)}"
end

def cmd_rm(names)
  resolve_names(names).each do |name|

      FileUtils.rm_rf(dir)
      puts "removed: #{name}"
    else
      warn "no such loop: #{name}"
    end
  end
end

def show_status(name)

    return
  end
  items = status['items'] || []
  puts "items (#{items.size}):"
  items.each do |it|
    line = "  - #{it['title'] || it['id']}"

    line += "  <#{it['status']}>" if it['status']

  end
  if (acts = status['action_items']) && !acts.empty?
    puts
    puts 'action items:'
    acts.each { |a| puts "  * #{a['text']}#{a['url'] ? "  #{a['url']}" : ''}" }
  end
  if (qs = status['open_questions']) && !qs.empty?
    puts
    puts 'open questions:'
    qs.each { |q| puts "  ? #{q['text']}#{q['url'] ? "  #{q['url']}" : ''}" }
  end

def render_by_task(all_status)
  grouped = Hash.new { |h, k| h[k] = { items: [], actions: [], questions: [] } }
  all_status.each do |name, s|
    next unless s
    (s['items'] || []).each         { |it| grouped[it['task'] || '_untagged'][:items]     << it.merge('_loop' => name) }
    (s['action_items'] || []).each  { |it| grouped[it['task'] || '_untagged'][:actions]   << it.merge('_loop' => name) }
    (s['open_questions'] || []).each { |it| grouped[it['task'] || '_untagged'][:questions] << it.merge('_loop' => name) }
  end
  return '' if grouped.empty?

  cards = grouped.sort_by { |k, _| k == '_untagged' ? 'zzz' : k }.map do |task, bucket|
    title = task == '_untagged' ? '(no task)' : task
    parts = []
    unless bucket[:items].empty?
      parts << '<h4>items</h4><ul>' + bucket[:items].map { |i|
        link = i['url'] ? "<a href='#{html_escape(i['url'])}'>#{html_escape(i['title'] || i['id'])}</a>" : html_escape(i['title'] || i['id'])
        "<li>#{link} <span class='src'>#{html_escape(i['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    unless bucket[:actions].empty?
      parts << '<h4>action items</h4><ul>' + bucket[:actions].map { |a|
        text = a['url'] ? "<a href='#{html_escape(a['url'])}'>#{html_escape(a['text'])}</a>" : html_escape(a['text'])
        "<li>#{text} <span class='src'>#{html_escape(a['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    unless bucket[:questions].empty?
      parts << '<h4>open questions</h4><ul>' + bucket[:questions].map { |q|
        text = q['url'] ? "<a href='#{html_escape(q['url'])}'>#{html_escape(q['text'])}</a>" : html_escape(q['text'])
        "<li>#{text} <span class='src'>#{html_escape(q['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    "<article class='task-card'><h3>#{html_escape(title)}</h3>#{parts.join}</article>"
  end.join

  "<section id='by-task'><header><h2>by task</h2></header>#{cards}</section>"
end

def tmux_capture(*args)

def session_exists?

def ensure_session

def window_exists?(window_id)
  return false if window_id.nil? || window_id.empty?
  out, ok = tmux_capture('list-windows', '-a', '-F', '#{window_id}')
  ok && out.split("\n").include?(window_id)
end

def kill_window(window_id)
  return unless window_id && !window_id.empty?
  system('tmux', 'kill-window', '-t', window_id, out: File::NULL, err: File::NULL)
end

def new_window(name)
  ensure_session
  # -P prints info about the new window; -F formats it. We want its window_id.
  window_id, _ = tmux_capture('new-window', '-d', '-t', "#{SESSION}:", '-n', name, '-P', '-F', '#{window_id}')
  pane_id, _   = tmux_capture('list-panes', '-t', window_id, '-F', '#{pane_id}')
  [window_id, pane_id.split("\n").first]
end

def render_by_task(all_status)
  grouped = Hash.new { |h, k| h[k] = { items: [], actions: [], questions: [] } }
  all_status.each do |name, s|
    next unless s
    (s['items'] || []).each         { |it| grouped[it['task'] || '_untagged'][:items]     << it.merge('_loop' => name) }
    (s['action_items'] || []).each  { |it| grouped[it['task'] || '_untagged'][:actions]   << it.merge('_loop' => name) }
    (s['open_questions'] || []).each { |it| grouped[it['task'] || '_untagged'][:questions] << it.merge('_loop' => name) }
  end
  return '' if grouped.empty?

  cards = grouped.sort_by { |k, _| k == '_untagged' ? 'zzz' : k }.map do |task, bucket|
    title = task == '_untagged' ? '(no task)' : task
    parts = []
    unless bucket[:items].empty?
      parts << '<h4>items</h4><ul>' + bucket[:items].map { |i|
        link = i['url'] ? "<a href='#{html_escape(i['url'])}'>#{html_escape(i['title'] || i['id'])}</a>" : html_escape(i['title'] || i['id'])
        "<li>#{link} <span class='src'>#{html_escape(i['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    unless bucket[:actions].empty?
      parts << '<h4>action items</h4><ul>' + bucket[:actions].map { |a|
        text = a['url'] ? "<a href='#{html_escape(a['url'])}'>#{html_escape(a['text'])}</a>" : html_escape(a['text'])
        "<li>#{text} <span class='src'>#{html_escape(a['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    unless bucket[:questions].empty?
      parts << '<h4>open questions</h4><ul>' + bucket[:questions].map { |q|
        text = q['url'] ? "<a href='#{html_escape(q['url'])}'>#{html_escape(q['text'])}</a>" : html_escape(q['text'])
        "<li>#{text} <span class='src'>#{html_escape(q['_loop'])}</span></li>"
      }.join + '</ul>'
    end
    "<article class='task-card'><h3>#{html_escape(title)}</h3>#{parts.join}</article>"
  end.join

  "<section id='by-task'><header><h2>by task</h2></header>#{cards}</section>"
end

def sync_codex(path:, servers:, authority:, opts:)
  text = File.exist?(path) ? File.read(path) : ''
  servers.each do |name, url|
    key = toml_key(name)
    header = /^\[mcp_servers\.#{Regexp.escape(key)}\]\s*$/
    lines = text.lines
    start = lines.index { |l| l =~ header }
    if start
      stop = ((start + 1)...lines.length).find { |i| lines[i] =~ /^\[/ } || lines.length
      block = lines[start...stop]
      if block.any? { |l| l =~ /^\s*command\s*=/ }
        warn "codex: '#{name}' is command-based; leaving untouched"
        next
      end
      if (i = block.index { |l| l =~ /^\s*url\s*=/ })
        block[i] = "url = #{url.to_json}\n"
      else
        block.insert(1, "url = #{url.to_json}\n")
      end
      lines[start...stop] = block
      text = lines.join
    else
      text += "\n[mcp_servers.#{key}]\nurl = #{url.to_json}\n"
    end
  end

  if opts.prune
    lines = text.lines
    ranges = []
    i = 0
    while i < lines.length
      if lines[i] =~ /^\[mcp_servers\.([^\]]+)\]\s*$/
        name = Regexp.last_match(1).delete('"')
        j = ((i + 1)...lines.length).find { |k| lines[k] =~ /^\[/ } || lines.length
        if !servers.key?(name) && lines[i...j].any? { |l| l =~ /^\s*url\s*=.*#{Regexp.escape(authority)}/ }
          ranges << (i...j)
          puts "  pruned codex server: #{name}"
        end
        i = j
      else
        i += 1
      end
    end
    ranges.reverse_each { |r| lines[r] = [] }
    text = lines.join
  end

  if opts.dry_run
    puts "[dry-run] #{path}"
    puts text
  elsif !File.exist?(path) || text != File.read(path)
    backup(path)
    File.write(path, text)
    puts "wrote #{path}"
  else
    puts "unchanged #{path}"
  end
end

def sync_json(path:, container_keys:, servers:, authority:, opts:, schema: nil, &entry_for)
  data = File.exist?(path) ? JSON.parse(File.read(path)) : {}
  data = { '$schema' => schema }.merge(data) if schema && !data.key?('$schema')
  container = container_keys.inject(data) { |h, k| h[k] ||= {} }

  before = container.keys.sort
  servers.each { |name, url| container[name] = entry_for.call(url) }
  removed = []
  if opts.prune
    container.each_key do |name|
      next if servers.key?(name)
      next unless registry_url?(container[name], authority)
      removed << name
      container.delete(name)
    end
  end
  after = container.keys.sort

  changed = (before != after) || servers.any? { |n, u| container[n] != entry_for.call(u) }
  if opts.dry_run
    puts "[dry-run] #{path}"
    puts JSON.pretty_generate(container)
  elsif changed || !File.exist?(path)
    FileUtils.mkdir_p(File.dirname(path))
    backup(path)
    File.write(path, JSON.pretty_generate(data) + "\n")
    puts "wrote #{path}"
  else
    puts "unchanged #{path}"
  end
  puts "  servers: #{after.join(', ')}"
  puts "  pruned: #{removed.join(', ')}" unless removed.empty?
end
