  def dir? = File.directory?(path)
  def file? = File.file?(path)
  def star? = path.include?('*')
  def files
    return absify(*Dir.glob(path)) if star?
    absify(*Dir.glob(glob))
  end

  def glob
    return path if star?
    return path.sub(/\/*$/, '/*') if dir?
    return '*' if path.strip.empty?
    path
  end

  def recent_files
    fs = files
    fs.sort_by do |f|
      File.mtime(f)
    rescue => e
      binding.pry
    end.reverse
  end

  def extensions = (files || []).filter_map{|f| x = File.extname(f).strip; next if x.empty?; x }.sort.uniq

  def absify(*files)
    (files.flatten.map(&File.method(:absolute_path)).sort || []).select{|f| File.file?(f) || File.directory?(f) }
  end
end
