def display_string(index = nil)
  num = index ? "#{(index + 1).to_s.rjust(3)}." : ""
  status = @enabled ? "[ON]" : "[OFF]"
  sound_str = @sound ? " (#{@sound})" : ""

  if one_time?
    if @trigger_at
      trigger_time = Time.parse(@trigger_at) rescue nil
      time_str = trigger_time ? trigger_time.strftime('%Y-%m-%d %H:%M') : @trigger_at
      "#{num} #{status} [ONCE @ #{time_str}] #{@message}#{sound_str}".strip
    else
      "#{num} #{status} [ONCE] #{@message}#{sound_str}".strip
    end
  else
    "#{num} #{status} [#{@interval}] #{@message}#{sound_str}".strip
  end
end

def parse_interval(str)
  return 3600 unless str # default 1 hour

  match = str.to_s.match(/^(\d+)([smhdw])$/i)
  return 3600 unless match

  num = match[1].to_i
  unit = match[2].downcase

  case unit
  when 's' then num
  when 'm' then num * 60
  when 'h' then num * 3600
  when 'd' then num * 86400
  when 'w' then num * 604800
  else 3600
  end
end

def should_trigger?
  return false unless @enabled

  if one_time?
    # One-time reminder: check if trigger_at time has passed
    return false if @last_triggered # already triggered
    return false unless @trigger_at

    trigger_time = Time.parse(@trigger_at) rescue nil
    return false unless trigger_time

    Time.now >= trigger_time
  else
    # Recurring reminder: check interval since last trigger
    return true if @last_triggered.nil?

    last_time = Time.parse(@last_triggered) rescue Time.at(0)
    Time.now - last_time >= interval_seconds
  end
end

def trigger!
  @last_triggered = Time.now.iso8601
  # For one-time reminders, disable after triggering
  @enabled = false if one_time?
