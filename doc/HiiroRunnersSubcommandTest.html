def test_subcommand_exact_match
  handler = -> { :ok }
  subcommand = Hiiro::Runners::Subcommand.new("h", "test", handler)

  assert subcommand.exact_match?("test")
  refute subcommand.exact_match?("tes")
  refute subcommand.exact_match?("testing")
end
