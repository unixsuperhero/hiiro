def initialize(root_dir: ENV["H_CONTEXT_DIR"] || File.join(Dir.home, "notes", "ctx"))
  @root_dir = File.expand_path(root_dir)
  @bundles_dir = File.join(@root_dir, "bundles")
  @named_bundles_dir = File.join(@bundles_dir, "named")
  @current_bundle_path = File.join(@bundles_dir, "current.bundle")
  ensure_dirs!
end

def append_paths(*paths)
  entries = bundle_entries
  normalized = paths.flatten.map { |path| normalize_existing_path(path) }
  write_bundle(entries + normalized)
  normalized
end

def ensure_dirs!
  FileUtils.mkdir_p(@root_dir)
  FileUtils.mkdir_p(@bundles_dir)
  FileUtils.mkdir_p(@named_bundles_dir)
  File.write(@current_bundle_path, "") unless File.exist?(@current_bundle_path)
end

def remove_entries(*targets)
  entries = bundle_entries
  to_remove = resolve_targets(entries, targets.flatten)
  remaining = entries.reject { |entry| to_remove.include?(entry) }
  write_bundle(remaining)
  to_remove
end

def write_bundle(entries)
  ensure_dirs!
  data = entries.uniq
  contents = data.empty? ? "" : data.join("\n") + "\n"
  File.write(@current_bundle_path, contents)
end
