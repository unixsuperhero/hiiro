def self.parse_time_arg(time_str)
  return [false, nil, '1h'] if time_str.nil?

  # Check for absolute time format: YYYYmmddHHMM
  if time_str =~ /^(\d{12})$/
    year = $1[0..3].to_i
    month = $1[4..5].to_i
    day = $1[6..7].to_i
    hour = $1[8..9].to_i
    min = $1[10..11].to_i
    trigger_at = Time.new(year, month, day, hour, min).iso8601
    return [true, trigger_at, nil]
  end

  # Check for relative interval format: 15m, 2h, etc.
  if time_str =~ /^(\d+)([smhdw])$/i
    num = $1.to_i
    unit = $2.downcase
    seconds = case unit
    when 's' then num
    when 'm' then num * 60
    when 'h' then num * 3600
    when 'd' then num * 86400
    when 'w' then num * 604800
    else num * 60
    end

    # This is a one-time reminder at a relative time
    trigger_at = (Time.now + seconds).iso8601
    return [true, trigger_at, time_str]
  end

  # Default: treat as recurring interval
  [false, nil, time_str]
end

def edit_reminders(reminders = nil)
  reminders_array = if reminders.nil?
    [REMINDER_TEMPLATE]
  elsif reminders.is_a?(Array)
    reminders.map { |r| r.is_a?(Reminder) ? r.to_h : r }
  else
    [reminders.is_a?(Reminder) ? reminders.to_h : reminders]
  end

  tmpfile = Tempfile.new(['reminder-edit-', '.yml'])
  tmpfile.write(reminders_array.to_yaml)
  tmpfile.close

  system(ENV['EDITOR'] || 'vim', tmpfile.path)

  updated_data = YAML.load_file(tmpfile.path)
  tmpfile.unlink

  updated_array = updated_data.is_a?(Array) ? updated_data : [updated_data]
  updated_array.map { |hash| Reminder.from_hash(hash) }
end

def ensure_reminders_file
  dir = File.dirname(@reminders_file)
  FileUtils.mkdir_p(dir) unless Dir.exist?(dir)
  File.write(@reminders_file, [].to_yaml) unless File.exist?(@reminders_file)
end

def find_reminder_by_ref(ref, reminders)
  if ref =~ /^\d+$/
    idx = ref.to_i - 1
    return [idx, reminders[idx]] if idx >= 0 && idx < reminders.length
  end
  [nil, nil]
end

def load_reminder_hash(reminders = nil)
  reminders ||= load_reminders
  reminders.each_with_index.each_with_object({}) do |(reminder, idx), h|
    h[reminder.display_string(idx)] = reminder
  end
end

def load_reminders
  ensure_reminders_file
  data = YAML.load_file(@reminders_file) || []
  data.map { |hash| Reminder.from_hash(hash) }
