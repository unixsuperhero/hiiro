def with_temp_file(content = "", extension = ".txt")
  Dir.mktmpdir do |dir|
    path = File.join(dir, "test#{extension}")
    File.write(path, content)
    yield path
  end
end

def with_yaml_file(data)
  require 'yaml'
  with_temp_dir do |dir|
    path = File.join(dir, "test.yml")
    File.write(path, YAML.dump(data))
    yield path, dir
  end
end
